Python Agent / 脚本调用 (标准视频接口)： 
import requests, time

BASE_URL = "http://127.0.0.1:8765/v1"
API_KEY = "sk-flow-你的密钥"

# 1. 提交视频生成任务
headers = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}
data = {
    "prompt": "一只可爱的小金毛幼犬在阳光下的绿色草坪上欢快奔跑，写实超清，4k",
    "model": "veo-3.1",
    "aspect_ratio": "16:9"
}
resp = requests.post(f"{BASE_URL}/video/generations", headers=headers, json=data).json()
task_id = resp["task_id"]
print("任务已提交, ID:", task_id)

# 2. 轮询获取出片结果
while True:
    status_resp = requests.get(f"{BASE_URL}/video/generations/{task_id}", headers=headers).json()
    if status_resp["status"] == "SUCCESS":
        print("出片成功！视频下载地址:", status_resp["download_url"])
        break
    elif status_resp["status"] == "FAILED":
        print("生成失败:", status_resp.get("error_message"))
        break
    time.sleep(3)
